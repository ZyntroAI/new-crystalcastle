scripts/github/submit_comment.py

from __future__ import annotations

import os
import subprocess
import sys


def run_gh(*args: str) -> str:
    result = subprocess.run(
        ["gh", *args],
        check=True,
        text=True,
        capture_output=True,
    )

    return result.stdout.strip()


def get_required_env(name: str) -> str:
    value = os.getenv(name)

    if not value:
        raise RuntimeError(
            f"Missing required environment variable: {name}"
        )

    return value


def submit_comment() -> None:
    repository = get_required_env("GITHUB_REPOSITORY")

    event_name = os.getenv(
        "GITHUB_EVENT_NAME",
        "unknown",
    )

    pr_number = os.getenv("PR_NUMBER")

    body = os.getenv("COMMENT_BODY")

    if not pr_number:
        raise RuntimeError(
            "PR_NUMBER is required"
        )

    if not body:
        raise RuntimeError(
            "COMMENT_BODY is required"
        )

    run_gh(
        "pr",
        "comment",
        pr_number,
        "--repo",
        repository,
        "--body",
        body,
    )

    print(
        f"Comment submitted to {repository}#{pr_number}"
    )
    print(f"Event: {event_name}")


if __name__ == "__main__":
    try:
        submit_comment()
    except subprocess.CalledProcessError as exc:
        print(
            exc.stderr,
            file=sys.stderr,
        )
        raise SystemExit(1) from exc

    except RuntimeError as exc:
        print(
            f"ERROR: {exc}",
            file=sys.stderr,
        )
        raise SystemExit(1) from exc

🇬🇧 GitHub Actions

.github/workflows/pr-auto-comment.yml

name: PR Auto Comment

on:
  workflow_run:
    workflows:
      - "CrystalCastle CodeRabbit + Tests"
    types:
      - completed

permissions:
  contents: read
  pull-requests: write

jobs:
  comment:
    name: Submit CI Comment
    runs-on: ubuntu-latest

    if: >
      github.event.workflow_run.event == 'pull_request'

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Resolve PR
        id: pr
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          REPOSITORY: ${{ github.repository }}
          HEAD_SHA: ${{ github.event.workflow_run.head_sha }}
        run: |
          PR_NUMBER=$(gh pr list \
            --repo "$REPOSITORY" \
            --search "$HEAD_SHA" \
            --json number \
            --jq '.[0].number')

          if [ -z "$PR_NUMBER" ]; then
            echo "No PR found."
            exit 1
          fi

          echo "number=$PR_NUMBER" >> "$GITHUB_OUTPUT"

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Submit comment
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          GITHUB_REPOSITORY: ${{ github.repository }}
          GITHUB_EVENT_NAME: ${{ github.event_name }}
          PR_NUMBER: ${{ steps.pr.outputs.number }}
          COMMENT_BODY: |
            ## 🤖 CrystalCastle CI Review

            The automated CI workflow has completed.

            | Gate | Status |
            |---|---|
            | CI | `${{ github.event.workflow_run.conclusion }}` |
            | CodeRabbit | Checked |
            | Required CI | Checked |

            **Workflow:** `${{ github.event.workflow_run.name }}`

            **Commit:** `${{ github.event.workflow_run.head_sha }}`

            Please review the workflow checks before merging.

        run: |
          python scripts/github/submit_comment.py

🔐 Permission

สำหรับ PR comment ใช้เพียง:

permissions:
  contents: read
  pull-requests: write

ไม่ต้องให้:

contents: write
actions: write
issues: write

ถ้า script มีหน้าที่แค่ comment บน PR

> หมายเหตุ: GitHub PR comments อยู่ใน Issues API model ดังนั้นบาง implementation ที่ใช้ gh issue comment จะต้องใช้ issues: write แทน แต่ถ้าใช้ gh pr comment ให้กำหนด pull-requests: write เป็นหลัก
