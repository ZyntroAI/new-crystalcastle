const jwt = require('jsonwebtoken');
const fs = require('fs');
const { Octokit } = require('octokit');
const db = require('./database');

// ============================================================
// ⚙️ CONFIG
// ============================================================
const APP_ID = process.env.GITHUB_APP_ID;
let PRIVATE_KEY;

if (process.env.GITHUB_PRIVATE_KEY_PATH) {
  PRIVATE_KEY = fs.readFileSync(process.env.GITHUB_PRIVATE_KEY_PATH, 'utf8');
} else {
  PRIVATE_KEY = process.env.GITHUB_PRIVATE_KEY.replace(/\\n/g, '\n');
}

if (!APP_ID || !PRIVATE_KEY) {
  console.error('❌ Missing GitHub App credentials');
  process.exit(1);
}

// ============================================================
// 🔑 JWT GENERATION (cached for performance)
// ============================================================
let cachedJWT = null;
let cachedJWTExpiry = 0;

function generateJWT() {
  const now = Math.floor(Date.now() / 1000);
  
  // Return cached JWT if still valid (refresh 60s before expiry)
  if (cachedJWT && now < cachedJWTExpiry - 60) {
    return cachedJWT;
  }
  
  const payload = {
    iat: now - 60,
    exp: now + 600,     // 10 minutes max
    iss: APP_ID
  };
  
  cachedJWT = jwt.sign(payload, PRIVATE_KEY, { algorithm: 'RS256' });
  cachedJWTExpiry = now + 600;
  
  return cachedJWT;
}

// ============================================================
// 🎟️ INSTALLATION TOKEN (with DB caching)
// ============================================================
async function getInstallationToken(installationId) {
  const now = new Date();
  
  // Check DB for valid cached token
  const installation = db.getInstallation(installationId);
  
  if (installation && installation.access_token && installation.token_expires_at) {
    const expiresAt = new Date(installation.token_expires_at);
    // Refresh if token expires in more than 60 seconds
    if (expiresAt.getTime() - now.getTime() > 60000) {
      return installation.access_token;
    }
  }
  
  // Token expired or not found — fetch new one
  console.log(`🔄 Fetching new installation token for ID: ${installationId}`);
  
  const appJWT = generateJWT();
  const octokit = new Octokit({ auth: appJWT });
  
  const response = await octokit.request(
    'POST /app/installations/{installation_id}/access_tokens',
    {
      installation_id: installationId,
      headers: { 'X-GitHub-Api-Version': '2022-11-28' }
    }
  );
  
  const { token, expires_at } = response.data;
  
  // Cache in DB
  if (installation) {
    db.updateInstallationToken(token, expires_at, installationId);
  }
  
  return token;
}

// ============================================================
// 🛠️ OCTOKIT FACTORY
// ============================================================
async function getInstallationOctokit(installationId) {
  const token = await getInstallationToken(installationId);
  return new Octokit({ 
    auth: token,
    throttle: {
      onRateLimit: (retryAfter, options) => {
        console.warn(`⚠️ Rate limit hit, retrying after ${retryAfter}s`);
        return true;
      },
      onSecondaryRateLimit: (retryAfter, options) => {
        console.warn(`⚠️ Secondary rate limit hit, retrying after ${retryAfter}s`);
        return true;
      }
    }
  });
}

function getAppOctokit() {
  return new Octokit({ 
    auth: generateJWT(),
    headers: { 'X-GitHub-Api-Version': '2022-11-28' }
  });
}

// ============================================================
// ✍️ SIGNATURE VERIFICATION
// ============================================================
const crypto = require('crypto');

function verifyWebhookSignature(rawBody, signatureHeader) {
  const secret = process.env.GITHUB_WEBHOOK_SECRET;
  if (!secret) return true;
  
  if (!signatureHeader) return false;
  
  const expectedSignature = `sha256=${crypto
    .createHmac('sha256', secret)
    .update(rawBody)
    .digest('hex')}`;
  
  try {
    return crypto.timingSafeEqual(
      Buffer.from(expectedSignature),
      Buffer.from(signatureHeader)
    );
  } catch {
    return false;
  }
}

// ============================================================
// 📤 EXPORTS
// ============================================================
module.exports = {
  generateJWT,
  getInstallationToken,
  getInstallationOctokit,
  getAppOctokit,
  verifyWebhookSignature,
  APP_ID
};
