const db = require('./database');
const auth = require('./githubAuth');

// Simple in-memory job scheduler (no external dependencies)
class Scheduler {
  constructor() {
    this.jobs = [];
    this.isRunning = false;
  }
  
  /**
   * Register a new job
   * @param {string} name - Job name
   * @param {number} intervalMs - Run interval in milliseconds
   * @param {Function} handler - Async function to execute
   * @param {boolean} runOnStart - Run immediately when scheduler starts
   */
  addJob(name, intervalMs, handler, runOnStart = false) {
    this.jobs.push({
      name,
      intervalMs,
      handler,
      runOnStart,
      lastRun: null,
      nextRun: runOnStart ? Date.now() : Date.now() + intervalMs,
      timeoutId: null
    });
    console.log(`📅 Scheduled job: ${name} (every ${this.formatInterval(intervalMs)})`);
  }
  
  formatInterval(ms) {
    const seconds = Math.floor(ms / 1000);
    const minutes = Math.floor(seconds / 60);
    const hours = Math.floor(minutes / 60);
    
    if (hours > 0) return `${hours}h`;
    if (minutes > 0) return `${minutes}m`;
    return `${seconds}s`;
  }
  
  async runJob(job) {
    try {
      console.log(`▶️ Running job: ${job.name}`);
      await job.handler();
      job.lastRun = new Date();
      console.log(`✅ Job completed: ${job.name}`);
    } catch (error) {
      console.error(`❌ Job failed: ${job.name}`, error.message);
    } finally {
      job.nextRun = Date.now() + job.intervalMs;
    }
  }
  
  start() {
    if (this.isRunning) return;
    this.isRunning = true;
    
    console.log('\n⏰ Scheduler started');
    console.log(`   Registered jobs: ${this.jobs.length}`);
    
    this.jobs.forEach(job => {
      if (job.runOnStart) {
        this.runJob(job);
      }
      
      const tick = () => {
        job.timeoutId = setTimeout(async () => {
          await this.runJob(job);
          tick();
        }, job.intervalMs);
      };
      
      // Schedule first run (adjust if runOnStart)
      const initialDelay = job.runOnStart ? job.intervalMs : job.intervalMs;
      job.timeoutId = setTimeout(async () => {
        if (!job.runOnStart) {
          await this.runJob(job);
        }
        tick();
      }, initialDelay);
    });
  }
  
  stop() {
    this.jobs.forEach(job => {
      if (job.timeoutId) {
        clearTimeout(job.timeoutId);
      }
    });
    this.isRunning = false;
    console.log('⏰ Scheduler stopped');
  }
  
  getStatus() {
    return this.jobs.map(job => ({
      name: job.name,
      interval: this.formatInterval(job.intervalMs),
      lastRun: job.lastRun?.toISOString() || null,
      nextRun: new Date(job.nextRun).toISOString()
    }));
  }
}

// 📋 JOB DEFINITIONS
// ============================================================
const scheduler = new Scheduler();

// Job 1: Refresh all installation tokens (every 50 minutes)
// GitHub tokens expire after 1 hour — refresh them proactively
scheduler.addJob('refresh-installation-tokens', 50 * 60 * 1000, async () => {
  const installations = db.getAllInstallations();
  console.log(`🔄 Refreshing tokens for ${installations.length} installations...`);
  
  for (const inst of installations) {
    try {
      await auth.getInstallationToken(inst.installation_id);
    } catch (error) {
      console.warn(`   ⚠️ Failed to refresh token for ${inst.account_login}:`, error.message);
    }
  }
}, false);

// Job 2: Daily repository health check (every 24 hours)
scheduler.addJob('daily-health-check', 24 * 60 * 60 * 1000, async () => {
  const installations = db.getAllInstallations();
  console.log(`🏥 Running daily health check across ${installations.length} installations...`);
  
  let totalRepos = 0;
  let totalIssues = 0;
  
  for (const inst of installations) {
    try {
      const octokit = await auth.getInstallationOctokit(inst.installation_id);
      
      // List accessible repositories
      const repos = await octokit.request('GET /installation/repositories', {
        per_page: 100,
        headers: { 'X-GitHub-Api-Version': '2022-11-28' }
      });
      
      totalRepos += repos.data.total_count;
      
      // Check each repo for open issues
      for (const repo of repos.data.repositories.slice(0, 10)) { // Limit to 10 per installation
        try {
          const issues = await octokit.request(
            'GET /repos/{owner}/{repo}/issues',
            {
              owner: repo.owner.login,
              repo: repo.name,
              state: 'open',
              per_page: 1,
              headers: { 'X-GitHub-Api-Version': '2022-11-28' }
            }
          );
          
          // Link header contains total count
          const linkHeader = issues.headers.link;
          if (linkHeader) {
            const match = linkHeader.match(/page=(\d+)>; rel="last"/);
            if (match) totalIssues += parseInt(match[1]);
          } else if (issues.data.length > 0) {
            totalIssues += 1;
          }
        } catch {
          // Skip repos where we can't access issues
        }
      }
      
    } catch (error) {
      console.warn(`   ⚠️ Health check failed for ${inst.account_login}:`, error.message);
    }
  }
  
  db.logAudit({
    action: 'daily_health_check',
    entity_type: 'system',
    entity_id: 'global',
    details: JSON.stringify({ installations: installations.length, repos: totalRepos, open_issues: totalIssues })
  });
  
  console.log(`🏥 Health check complete: ${totalRepos} repos, ${totalIssues} open issues`);
  
}, false);

// Job 3: Hourly stats summary (every 1 hour)
scheduler.addJob('hourly-stats-summary', 60 * 60 * 1000, async () => {
  const stats = db.getStats();
  console.log(`📊 Hourly Stats — Installations: ${stats.installations_count}, ` +
              `Users: ${stats.users_count}, Webhooks: ${stats.webhooks_processed}`);
}, true);

// Job 4: Weekly cleanup of old webhook logs (every 7 days)
scheduler.addJob('weekly-log-cleanup', 7 * 24 * 60 * 60 * 1000, async () => {
  // Delete webhook logs older than 30 days
  const thirtyDaysAgo = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString();
  
  const result = db.db.prepare(
    'DELETE FROM webhook_logs WHERE created_at < ?'
  ).run(thirtyDaysAgo);
  
  console.log(`🧹 Cleaned up ${result.changes} old webhook log entries`);
  
  db.logAudit({
    action: 'log_cleanup',
    entity_type: 'system',
    entity_id: 'webhook_logs',
    details: JSON.stringify({ deleted: result.changes, older_than: thirtyDaysAgo })
  });
}, false);

// ============================================================
// 📤 EXPORTS
// ============================================================
module.exports = scheduler;
