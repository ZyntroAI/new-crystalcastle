CREATE TABLE whatsapp_responses (
  id SERIAL PRIMARY KEY,
  user_number TEXT NOT NULL,
  reply_id TEXT NOT NULL,
  pr_number INT,
  timestamp TIMESTAMPTZ DEFAULT now()
);

CREATE OR REPLACE FUNCTION notify_whatsapp_response()
RETURNS trigger AS $$
BEGIN
  PERFORM pg_notify(
    'whatsapp_channel',
    json_build_object(
      'id', NEW.id,
      'user_number', NEW.user_number,
      'reply_id', NEW.reply_id,
      'pr_number', NEW.pr_number,
      'timestamp', NEW.timestamp
    )::text
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER whatsapp_response_trigger
AFTER INSERT ON whatsapp_responses
FOR EACH ROW EXECUTE FUNCTION notify_whatsapp_response();
